    sig/.style      = {draw=navy,   line width=0.8pt, ->},
    redsig/.style   = {draw=tlgred, line width=1.2pt, ->},

  % ---------------- Wishbone slave + cfg regfile (top row) -------------

    node[wlbl, midway, above] {cfg};

  \node[tlg, minimum width=14mm, minimum height=11mm, anchor=west]
    (xnor) at ($(xreg.east)+(0.55,0)$) {XNOR\\\scriptsize 64-bit};
  \node[tlg, minimum width=22mm, minimum height=11mm, anchor=west]
    (pc)   at ($(xnor.east)+(0.55,0)$)
    {Popcount tree\\\scriptsize 11 chunks $\!\to\!$ 7-b};
  \node[tlg, minimum width=15mm, minimum height=11mm, anchor=west]
    (cmp)  at ($(pc.east)+(0.55,0)$)
    {$\geq \theta_i$\\\scriptsize compare};
  \node[secondary, minimum width=15mm, minimum height=11mm, anchor=west]
    (yreg) at ($(cmp.east)+(0.55,0)$) {$y$ reg\\\scriptsize 16-bit};

  \draw[sig]    (xreg.east) -- (xnor.west) node[wlbl, midway, above] {64};
  \draw[redsig] (xnor.east) -- (pc.west);
  \draw[redsig] (pc.east)   -- (cmp.west)
